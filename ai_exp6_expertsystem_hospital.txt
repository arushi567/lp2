# expert system: hospital & medical facilities

# knowledge base (stores rules: symptom -> details)
knowledge_base = {
    "fever": {
        "department": "General Physician",
        "severity": "Low",
        "advice": "Take rest and basic medication",
        "reason": "Fever usually indicates infection"
    },
    "cough": {
        "department": "General Physician",
        "severity": "Low",
        "advice": "Check for cold or respiratory infection",
        "reason": "Cough is related to respiratory issues"
    },
    "chest pain": {
        "department": "Cardiologist",
        "severity": "High",
        "advice": "Visit a multi-speciality hospital immediately",
        "reason": "Chest pain may indicate heart problems"
    },
    "fracture": {
        "department": "Orthopedic",
        "severity": "Medium",
        "advice": "Get X-ray and proper treatment",
        "reason": "Bone injury requires orthopedic care"
    },
    "rash": {
        "department": "Dermatologist",
        "severity": "Low",
        "advice": "Consult skin specialist",
        "reason": "Skin issues require dermatology care"
    },
    "pregnancy": {
        "department": "Gynecologist",
        "severity": "Medium",
        "advice": "Regular checkups required",
        "reason": "Pregnancy needs specialized care"
    },
    "accident": {
        "department": "Emergency",
        "severity": "High",
        "advice": "Immediate emergency care required",
        "reason": "Accidents require urgent treatment"
    },
    "bleeding": {
        "department": "Emergency",
        "severity": "High",
        "advice": "Control bleeding and rush to hospital",
        "reason": "Excess bleeding is life-threatening"
    },
    "headache": {
        "department": "Neurologist / General Physician",
        "severity": "Low",
        "advice": "Check stress or migraine",
        "reason": "Headache may be neurological or general"
    },
    "stomach pain": {
        "department": "Gastroenterologist",
        "severity": "Medium",
        "advice": "Avoid spicy food and consult doctor",
        "reason": "Digestive system issue"
    }
}


def diagnose(symptoms):
    # stores matched results
    results = []
    # stores explanations (why decision was made)
    explanations = []

    # inference engine (rule checking)
    for symptom in symptoms:
        if symptom in knowledge_base:
            rule = knowledge_base[symptom]

            # store result
            results.append({
                "symptom": symptom,
                "department": rule["department"],
                "severity": rule["severity"],
                "advice": rule["advice"]
            })

            # explanation for decision
            explanations.append(f"For '{symptom}': {rule['reason']}")
        else:
            explanations.append(
                f"For '{symptom}': No rule found, consult General Physician"
            )

    return results, explanations


def display_results(results, explanations):
    print("\n" + "="*40)
    print("        diagnosis result")
    print("="*40)

    if not results:
        print("no specific condition identified.")
        return

    # to store unique departments
    departments = set()

    # track highest severity
    highest_severity = "Low"
    severity_order = {"Low": 1, "Medium": 2, "High": 3}

    for res in results:
        departments.add(res["department"])

        if severity_order[res["severity"]] > severity_order[highest_severity]:
            highest_severity = res["severity"]

    # display departments
    print("\nrecommended department(s):")
    for dept in departments:
        print("-", dept)

    # display severity
    print("\nseverity level:", highest_severity)

    # display advice
    print("\nadvice:")
    for res in results:
        print(f"- ({res['symptom']}): {res['advice']}")

    # display explanation (why)
    print("\nexplanation (why?):")
    for exp in explanations:
        print("-", exp)


def main():
    print("="*45)
    print("   hospital expert system (ai based)")
    print("="*45)

    print("\navailable symptoms:")
    for symptom in knowledge_base.keys():
        print("-", symptom)

    # user input
    user_input = input("\nenter symptoms (comma separated): ")

    # convert input to list
    symptoms = [s.strip().lower() for s in user_input.split(",")]

    # run expert system
    results, explanations = diagnose(symptoms)

    # display output
    display_results(results, explanations)


# run program
if __name__ == "__main__":
    main()
