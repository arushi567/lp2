# 🧱 STEP 5 — CREATE APEX CLASS (BACKEND LOGIC)

Go to:

```
File → New → Apex Class
```

Name: TaskManagerController


Paste your code (NO CHANGES):
-----------------------------------------------------------------------------------------------------------------------
-----------------------------------------------------------------------------------------------------------------------

public class TaskManagerController {
    
    public Task newTask { get; set; }
    public List<Task> taskList { get; set; }
    public String selectedTaskId { get; set; }

    public TaskManagerController() {
        newTask = new Task(Status = 'Not Started');
        loadTasks();
    }

    public void loadTasks() {
        taskList = [SELECT Id, Subject, Status FROM Task ORDER BY CreatedDate DESC LIMIT 50];
    }

    public void addTask() {
        newTask.Status = 'Not Started'; 
        insert newTask;
        newTask = new Task(Status = 'Not Started');
        loadTasks();
    }

    public void markDone() {
        update new Task(Id = selectedTaskId, Status = 'Completed');
        loadTasks();
    }

    public void deleteTask() {
        delete new Task(Id = selectedTaskId);
        loadTasks();
    }
}

# 🌐 STEP 6 — CREATE VISUALFORCE PAGE (FRONTEND UI)

Go to:

Setup → Visualforce Pages → New


Name: TaskManagerPage


Paste your code:

----------------------------------------------------------------------------------------------------------------------
----------------------------------------------------------------------------------------------------------------------

<apex:page controller="TaskManagerController" docType="html-5.0">
    <apex:form id="mainForm">
        <apex:pageBlock title="Task Manager App">
            
            <apex:pageBlockSection title="Add New Task" columns="2">
                <apex:pageBlockSectionItem>
                    <apex:outputLabel value="What needs to be done?" />
                    <apex:inputText value="{!newTask.Subject}" required="true"/>
                </apex:pageBlockSectionItem>
                <apex:commandButton value="Add Task" action="{!addTask}" reRender="mainForm" />
            </apex:pageBlockSection>
            
            <apex:actionRegion>
                <apex:pageBlockSection title="Your Tasks" columns="1">
                    <apex:pageBlockTable value="{!taskList}" var="t">
                        <apex:column headerValue="Actions">
                            <apex:commandLink value="Mark Done" action="{!markDone}" reRender="mainForm" rendered="{!t.Status != 'Completed'}">
                                <apex:param name="taskId" value="{!t.Id}" assignTo="{!selectedTaskId}" />
                            </apex:commandLink>
                            <apex:outputText value="Done!" rendered="{!t.Status == 'Completed'}" style="color:green; font-weight:bold;"/>
                            &nbsp;|&nbsp;
                            <apex:commandLink value="Delete" action="{!deleteTask}" reRender="mainForm" style="color:red;">
                                <apex:param name="taskId" value="{!t.Id}" assignTo="{!selectedTaskId}" />
                            </apex:commandLink>
                        </apex:column>
                        <apex:column value="{!t.Subject}" headerValue="Task" />
                        <apex:column value="{!t.Status}" />
                    </apex:pageBlockTable>
                </apex:pageBlockSection>
            </apex:actionRegion>
            
        </apex:pageBlock>
    </apex:form>
</apex:page>

// 1. "Open" the page by creating a new controller instance
TaskManagerController controller = new TaskManagerController();

// 2. Mimic typing a subject into the text box and clicking "Add Task"
controller.newTask.Subject = 'Learn Execute Anonymous';
controller.addTask();

System.debug('--- AFTER ADD ---');
for (Task t : controller.taskList) {
    System.debug('Task: ' + t.Subject + ' | Status: ' + t.Status);
}

// 3. Get the ID of the task we just created
Id latestTaskId = controller.taskList[0].Id;

// 4. Mimic clicking the "Mark Done" link
controller.selectedTaskId = latestTaskId;
controller.markDone();

System.debug('--- AFTER MARK DONE ---');
for (Task t : controller.taskList) {
    System.debug('Task: ' + t.Subject + ' | Status: ' + t.Status);
}

// 5. Mimic clicking the "Delete" link
controller.selectedTaskId = latestTaskId;
controller.deleteTask();

System.debug('--- AFTER DELETE ---');
System.debug('Remaining Tasks: ' + controller.taskList.size());
